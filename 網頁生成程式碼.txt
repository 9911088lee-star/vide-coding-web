        // Price Calculator
        function calculatePrice() {
            const courseSelect = document.getElementById('form-course-select');
            const qtySelect = document.getElementById('form-quantity');
            const priceDisplay = document.getElementById('total-price-display');
            const discountHint = document.getElementById('discount-hint');

            const selectedOption = courseSelect.options[courseSelect.selectedIndex];
            const basePrice = selectedOption.dataset.price ? parseInt(selectedOption.dataset.price) : 0;
            const qty = parseInt(qtySelect.value);

            if (basePrice === 0) {
                priceDisplay.textContent = 'NT$ 0';
                discountHint.textContent = '請選擇課程計算金額';
                return;
            }

            let discount = 1.0;
            let hintText = '單人正價';

            if (qty === 2) {
                discount = 0.95;
                hintText = '雙人同行 95 折優惠';
            } else if (qty >= 3) {
                discount = 0.9;
                hintText = '團體報名 9 折優惠';
            }

            const total = Math.round(basePrice * qty * discount);
            priceDisplay.textContent = `NT$ ${total.toLocaleString()}`;
            discountHint.textContent = hintText;
        }

        // Reusable Modal Functions
        function showModal(htmlContent) {
            const modal = document.getElementById('modal-overlay');
            const body = document.getElementById('modal-body');
            body.innerHTML = htmlContent;
            modal.classList.remove('hidden');
            setTimeout(() => {
                modal.classList.remove('opacity-0');
                modal.children[0].classList.replace('scale-95', 'scale-100');
            }, 10);
        }

        function closeModal() {
            const modal = document.getElementById('modal-overlay');
            modal.children[0].classList.replace('scale-100', 'scale-95');
            modal.classList.add('opacity-0');
            setTimeout(() => {
                modal.classList.add('hidden');
            }, 300);
        }

        // Toast Notification System
        function showToast(message, type = 'info') {
            const container = document.getElementById('toast-container');
            const toast = document.createElement('div');
            const bgClass = type === 'success' ? 'bg-leaf-800 text-white border-leaf-600' : 'bg-red-800 text-white border-red-600';
            const icon = type === 'success' ? 'fa-circle-check text-amber-400' : 'fa-circle-exclamation';

            toast.className = `pointer-events-auto px-4 py-3 rounded-xl border shadow-xl flex items-center gap-3 text-xs font-semibold ${bgClass} transition-all duration-300 transform translate-y-2 opacity-0`;
            toast.innerHTML = `<i class="fa-solid ${icon} text-base"></i><span>${message}</span>`;
            
            container.appendChild(toast);
            setTimeout(() => {
                toast.classList.remove('translate-y-2', 'opacity-0');
            }, 10);

            setTimeout(() => {
                toast.classList.add('opacity-0', 'translate-y-2');
                setTimeout(() => toast.remove(), 300);
            }, 3500);
        }
